# PremiumPad: reads the controllers' Home button for Premium Games' console mode, in the background (also while a game
# in its own program has the focus), and the real Xbox / PS buttons. Prints one line per event on stdout.
cmake_minimum_required(VERSION 3.24)
project(PremiumPad C)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
foreach(o AUDIO VIDEO RENDER SENSOR POWER HAPTIC FILESYSTEM LOCALE MISC)
  set(SDL_${o} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
set(SDL_EVENTS ON CACHE BOOL "" FORCE)
FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.30.9 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL2)
add_executable(PremiumPad WIN32 padwatch.c)
target_link_libraries(PremiumPad PRIVATE SDL2::SDL2-static)
if(MSVC)
  set_property(TARGET PremiumPad PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  target_link_options(PremiumPad PRIVATE /SUBSYSTEM:CONSOLE /ENTRY:mainCRTStartup)
endif()
